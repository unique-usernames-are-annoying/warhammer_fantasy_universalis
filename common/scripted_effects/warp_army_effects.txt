take_and_spawn_army_at_province = {
	from = {
		set_variable = {
			which = infantry_units_to_warp
			value = 0
		}
		set_variable = {
			which = cavalry_units_to_warp
			value = 0
		}
		set_variable = {
			which = artillery_units_to_warp
			value = 0
		}
		
		capital_scope = {
			## registering inf
			while = {
				limit = {
					infantry_in_province = OWNER
				}
				owner = {
					change_variable = {
						which = infantry_units_to_warp 
						value = 1
					}
				}
				kill_units = {
					who = OWNER
					type = infantry
					amount = 1
				}
			}

			## registering cav
			while = {
				limit = {
					cavalry_in_province = OWNER
				}
				owner = {
					change_variable = {
						which = cavalry_units_to_warp  
						value = 1
					}
				}
				kill_units = {
					who = OWNER
					type = cavarly
					amount = 1
				}
			}

			## registering arty
			while = {
				limit = {
					artillery_in_province = OWNER
				}
				owner = {
					change_variable = {
						which = artillery_units_to_warp   
						value = 1
					}
				}
				kill_units = {
					who = OWNER
					type = artillery
					amount = 1
				}
			}

			
		}
		## spawn inf
		while = {
			limit = {
					check_variable = {
						which = infantry_units_to_warp
						value = 1
					}
			}
			
			prev = {
				infantry = PREV
			}

			subtract_variable = {
				which = infantry_units_to_warp 
				value = 1
			}
		}

		## spawn cav
		while = {
			limit = {
				check_variable = {
					which = cavalry_units_to_warp 
					value = 1
				}
				
			}
			
			prev = {
				cavalry = PREV
			}

			subtract_variable = {
				which = cavalry_units_to_warp 
				value = 1
			}
		}
		## spawn arty
		while = {
			limit = {
				check_variable = {
					which = artillery_units_to_warp 
					value = 1
				}

			}
			
			prev = {
				artillery = PREV
			}

			subtract_variable = {
				which = artillery_units_to_warp 
				value = 1
			}
		}

	}

}
