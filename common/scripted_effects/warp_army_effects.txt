take_and_spawn_army_at_province = {
	from = {
		export_to_variable = {
			which = max_units_to_warp
			value = land_forcelimit
		}
		# set_variable = {
		# 	which = max_units_to_warp
		# 	which = forcelimit
		# }
		multiply_variable = {
			which = max_units_to_warp
			value = 0.5
		}

		log = "[From.GetName] max_units_to_warp after divide = [From.max_units_to_warp.GetValue]"

        set_variable = { which = infantry_units_to_warp value = 0 }
        set_variable = { which = cavalry_units_to_warp  value = 0 }
        set_variable = { which = artillery_units_to_warp value = 0 }
		
		capital_scope = {
			## registering inf
			while = {
				limit = {
					## if there are units in the capital and there's still room in max_units_to_warp
					AND = {
						infantry_in_province = OWNER
						owner = {
							check_variable = {
								which = max_units_to_warp
								value = 1
							}
						}
					}
				}
				## increase inf by one and remove a slot from max_units_to_warp
				owner = {
					change_variable = {
						which = infantry_units_to_warp 
						value = 1
					}
					subtract_variable = {
						which = max_units_to_warp
						value = 1
					}
				}
				kill_units = {
					who = OWNER
					type = infantry
					amount = 1
				}
			}

			## registering cav
			while = {
				limit = {
					AND = {
						cavalry_in_province = OWNER
						owner = {
							check_variable = {
								which = max_units_to_warp
								value = 1
							}
						}
					}
				}
				owner = {
					change_variable = {
						which = cavalry_units_to_warp  
						value = 1
					}
					subtract_variable = {
						which = max_units_to_warp
						value = 1
					}
				}
				kill_units = {
					who = OWNER
					type = cavalry
					amount = 1
				}
			}

			## registering arty
			while = {
				limit = {
					AND = {
						artillery_in_province = OWNER
						owner = {
							check_variable = {
								which = max_units_to_warp
								value = 1
							}
						}
					}
				}
				owner = {
					change_variable = {
						which = artillery_units_to_warp   
						value = 1
					}
					subtract_variable = {
						which = max_units_to_warp
						value = 1
					}
				}
				kill_units = {
					who = OWNER
					type = artillery
					amount = 1
				}
			}

			
		}
		## spawn inf
		while = {
			limit = {
					check_variable = {
						which = infantry_units_to_warp
						value = 1
					}
			}
			## prev here is the province
			prev = {
				## prev here is FROM
				infantry = PREV
			}

			subtract_variable = {
				which = infantry_units_to_warp 
				value = 1
			}
		}

		## spawn cav
		while = {
			limit = {
				check_variable = {
					which = cavalry_units_to_warp 
					value = 1
				}
				
			}
			
			prev = {
				cavalry = PREV
			}

			subtract_variable = {
				which = cavalry_units_to_warp 
				value = 1
			}
		}
		## spawn arty
		while = {
			limit = {
				check_variable = {
					which = artillery_units_to_warp 
					value = 1
				}

			}
			
			prev = {
				artillery = PREV
			}

			subtract_variable = {
				which = artillery_units_to_warp 
				value = 1
			}
		}

	}

}
